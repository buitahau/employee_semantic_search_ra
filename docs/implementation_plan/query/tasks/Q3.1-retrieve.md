## Section C — `query/retrieve.py`

### SQL

Two variants depending on whether `QueryAnalysis.filter` is empty:

**Without filter** (empty dict):
```sql
SELECT
    employee_id,
    field_type,
    chunk_text,
    metadata,
    1 - (embedding <=> %s::vector) AS score
FROM skills_search_index
ORDER BY score DESC
LIMIT 100;
```

**With filter** (non-empty dict):
```sql
SELECT
    employee_id,
    field_type,
    chunk_text,
    metadata,
    1 - (embedding <=> %s::vector) AS score
FROM skills_search_index
WHERE metadata @> %s::jsonb
ORDER BY score DESC
LIMIT 100;
```

- `%s` placeholders only — no string interpolation
- Vector is passed as a JSON-serialised string: `"[0.12, -0.34, ...]"` — psycopg2 binds it as text and `::vector` casts it inside the DB
- Filter is passed as a JSON-serialised string: `json.dumps(filter_dict)` — `::jsonb` cast handles the conversion

### Implementation

```python
import json
import logging

import psycopg2.extras

from common.db import _vector_connect
from common.embedding import embed
from query.types import ChunkHit, QueryAnalysis

_logger = logging.getLogger(__name__)

_SQL_BASE = """
    SELECT
        employee_id,
        field_type,
        chunk_text,
        metadata,
        1 - (embedding <=> %s::vector) AS score
    FROM skills_search_index
    {where}
    ORDER BY score DESC
    LIMIT 100
"""


def retrieve(analysis: QueryAnalysis) -> list[ChunkHit]:
    vector = embed(analysis.query)
    vector_str = json.dumps(vector)

    if analysis.filter:
        sql = _SQL_BASE.format(where="WHERE metadata @> %s::jsonb")
        params = (vector_str, json.dumps(analysis.filter))
    else:
        sql = _SQL_BASE.format(where="")
        params = (vector_str,)

    with _vector_connect() as conn:
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(sql, params)
            rows = cur.fetchall()

    return [
        ChunkHit(
            employee_id=row["employee_id"],
            field_type=row["field_type"],
            chunk_text=row["chunk_text"],
            metadata=dict(row["metadata"]),
            score=float(row["score"]),
        )
        for row in rows
    ]
```

**Notes:**
- `_vector_connect()` connects to the vector DB (`VECTOR_DB_*`) — not the source DB
- The cursor is opened inside the connection context; `_vector_connect()` yields the raw `conn`, not a cursor
- `embed()` failure propagates as an exception — the caller (Q6.1 endpoint) maps this to HTTP 500 per Q7.2
- Results are already ordered by `score DESC` by the DB; no Python-side sort needed

## Acceptance Tests

```python
import json
from unittest.mock import MagicMock, call, patch

from query.retrieve import retrieve
from query.types import ChunkHit, QueryAnalysis


def _make_analysis(filter_dict=None):
    return QueryAnalysis(
        intent="find_employees",
        query="java backend experience",
        heuristic="rank employees by score descending",
        filter=filter_dict or {},
    )


def _make_row(employee_id=1, score=0.9):
    return {
        "employee_id": employee_id,
        "field_type": "experiences",
        "chunk_text": "Experience: java backend",
        "metadata": {"skills": ["Java"]},
        "score": score,
    }


@patch("query.retrieve.embed", return_value=[0.1] * 384)
@patch("query.retrieve._vector_connect")
def test_no_filter_omits_where_clause(mock_connect, mock_embed):
    mock_cur = MagicMock()
    mock_cur.fetchall.return_value = [_make_row()]
    mock_connect.return_value.__enter__.return_value.cursor.return_value.__enter__.return_value = mock_cur

    result = retrieve(_make_analysis())

    executed_sql = mock_cur.execute.call_args[0][0]
    assert "WHERE" not in executed_sql
    assert len(result) == 1
    assert isinstance(result[0], ChunkHit)


@patch("query.retrieve.embed", return_value=[0.1] * 384)
@patch("query.retrieve._vector_connect")
def test_filter_adds_where_clause(mock_connect, mock_embed):
    mock_cur = MagicMock()
    mock_cur.fetchall.return_value = [_make_row()]
    mock_connect.return_value.__enter__.return_value.cursor.return_value.__enter__.return_value = mock_cur

    result = retrieve(_make_analysis(filter_dict={"skills": ["Java"]}))

    executed_sql = mock_cur.execute.call_args[0][0]
    assert "WHERE" in executed_sql
    assert "metadata @>" in executed_sql


@patch("query.retrieve.embed", side_effect=RuntimeError("ONNX failure"))
@patch("query.retrieve._vector_connect")
```
